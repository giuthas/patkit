 53class PatAnnotation(ABC):
 54    """
 55    Base class for Textgrid Point and Interval to enable editing with GUI.
 56    """
 57
 58    def __init__(
 59            self,
 60            time: float,
 61            label: None | Transcript,
 62    ) -> None:
 63        self._time = time
 64        self.label = label
 65
 66    @property
 67    def time(self) -> float:
 68        """Location of this Point."""
 69        return self._time
 70
 71    @time.setter
 72    def time(self, time: float) -> None:
 73        self._time = time
 74
 75    @abstractmethod
 76    def contains(self, time: float, epsilon: float | None) -> bool:
 77        """
 78        Does this Interval contain `time` or is this Point at `time`.
 79
 80        'Being at time' is defined in the sense of 'within epsilon of time'.
 81
 82        Parameters
 83        ----------
 84        time : float
 85            The time in seconds to test against this Annotation.
 86        epsilon : float | None
 87            The precision (in seconds) to use in comparisons. The default value
 88            None will result in PATKIT_EPSILON, being used. For expected
 89            behaviour, `configuration.data_config.epsilon` should be passed
 90            here.
 91
 92        Returns
 93        -------
 94        bool
 95            True if `time` is in this Interval or at this Point.
 96        """

 99class PatPoint(PatAnnotation):
100    """TextGrid Point representation to enable editing with GUI."""

102    @classmethod
103    def from_textgrid_point(
104            cls,
105            point: Point,
106    ) -> Self:
107        """
108        Copy the info of a Python TextGrids Interval into a new PatInterval.
109
110        Only xmin and text are copied from the original Interval. xmax is
111        assumed to be handled by either the next PatInterval or the
112        constructing method if this is the last Interval.
113
114        Since PatIntervals are doubly linked, an attempt will be made to link
115        prev and next to this interval.
116
117        Returns the newly created PatInterval.
118        """
119        return cls(
120            time=point.xpos,
121            label=point.text,
122        )

124    def __init__(
125            self,
126            time: float,
127            label: None | Transcript
128    ) -> None:
129        super().__init__(time=time, label=label)
130
131    def contains(self, time: float, epsilon: float | None) -> bool:
132        if epsilon is None:
133            epsilon = PATKIT_EPSILON
134        if self._time - epsilon < time < self._time + epsilon:
135            return True
136        return False

139class PatInterval(PatAnnotation):
140    """TextGrid Interval representation to enable editing with GUI."""
141
142    @classmethod
143    def from_textgrid_interval(
144        cls,
145        interval: Interval,
146        prev_interval: None | Self,
147        next_interval: None | Self = None
148    ) -> Self:
149        """
150        Copy the info of a Python TextGrids Interval into a new PatInterval.
151
152        Only xmin and text are copied from the original Interval. xmax is
153        assumed to be handled by either the next PatInterval or the
154        constructing method if this is the last Interval.
155
156        Since PatIntervals are doubly linked, an attempt will be made to link
157        prev and next to this interval.
158
159        Returns the newly created PatInterval.
160        """
161        return cls(
162            begin=interval.xmin,
163            label=interval.text,
164            prev_interval=prev_interval,
165            next_interval=next_interval)
166
167    def __init__(self,
168                 begin: float,
169                 label: None | Transcript,
170                 prev_interval: None | Self = None,
171                 next_interval: None | Self = None) -> None:
172        super().__init__(
173            time=begin,
174            label=label,
175        )
176
177        self._prev_interval = prev_interval
178        if self.prev:
179            self.prev._next_interval = self
180
181        self._next_interval = next_interval
182        if self.next:
183            self.next.prev = self
184
185    def __repr__(self) -> str:
186        return (f"{self.__class__.__name__}: text: '{self.label}'\t "
187                f"begin: {self.begin}, end: {self.end}")
188
189    @property
190    def next(self) -> Self | None:
191        """The next annotation, if any."""
192        return self._next_interval
193
194    @next.setter
195    def next(self, next_interval: Self | None) -> None:
196        if next_interval != self._next_interval:
197            self._next_interval = next_interval
198
199    @property
200    def prev(self) -> Self | None:
201        """The previous annotation, if any."""
202        return self._prev_interval
203
204    @prev.setter
205    def prev(self, prev_interval: Self | None) -> None:
206        if prev_interval != self._prev_interval:
207            self._prev_interval = prev_interval
208
209    @property
210    def begin(self) -> float:
211        """Beginning time point of the interval."""
212        return self._time
213
214    @begin.setter
215    def begin(self, value: float) -> None:
216        self._time = value
217
218    @property
219    def mid(self) -> float | None:
220        """
221        Middle time point of the interval.
222
223        This is a property that will return None
224        if this Interval is the one that marks
225        the last boundary.
226        """
227        if self._next_interval:
228            return (self.begin + self._next_interval.begin)/2
229        return None
230
231    @property
232    def end(self) -> float | None:
233        """
234        End time point of the interval.
235
236        This is a property that will return None
237        if this Interval is the one that marks
238        the last boundary.
239        """
240        if self._next_interval:
241            return self._next_interval.begin
242        return None
243
244    @end.setter
245    def end(self, value: float) -> None:
246        if self._next_interval:
247            self._next_interval.begin = value
248
249    def is_at_time(self, time: float, epsilon) -> bool:
250        """
251        Intervals are considered equivalent if the difference between their
252        `begin` values is < epsilon. Epsilon is a constant defined in patkit's
253        configuration.
254        """
255        return abs(self.begin - time) < epsilon
256
257    def is_last(self) -> bool:
258        """Is this the last Interval in this Tier."""
259        return self._next_interval is None
260
261    def is_legal_value(self, time: float, epsilon: float) -> bool:
262        """
263        Check if the given time is between the previous and next boundary.
264
265        Usual caveats about float testing don't apply, because each boundary is
266        padded with patkit epsilon. Tests used do not include equality with
267        either bounding boundary, and that may or may not be trusted to be the
268        actual case depending on how small the epsilon is.
269
270        Returns True, if time is  between the previous and next boundary.
271        """
272        return (time + epsilon < self._next_interval.begin and
273                time > epsilon + self.prev.begin)
274
275    def contains(self, time: float, epsilon: float | None) -> bool:
276        if self.begin < time < self.end:
277            return True
278        return False

281class PatTier(list):
282    """TextGrid Tier representation to enable editing with GUI."""
283
284    @classmethod
285    def from_textgrid_tier(cls, tier: Tier) -> Self:
286        """
287        Copy a Python TextGrids Tier as a PatTier.
288
289        Returns the newly created PatTier.
290        """
291        return cls(tier)
292
293    def __init__(self, tier: Tier) -> None:
294        super().__init__()
295        last_interval = None
296        prev = None
297        for interval in tier:
298            current = PatInterval.from_textgrid_interval(interval, prev)
299            self.append(current)
300            prev = current
301            last_interval = interval
302        self.append(PatInterval(last_interval.xmax, None, prev))
303
304    def __repr__(self) -> str:
305        representation = f"{self.__class__.__name__}:\n"
306        for interval in self:
307            representation += str(interval) + "\n"
308        return representation
309
310    @property
311    def begin(self) -> float:
312        """
313        Begin timestamp.
314
315        Corresponds to a TextGrid Interval's xmin.
316
317        This is a property and the actual value is generated from the first
318        PatInterval of this PatTier.
319        """
320        return self[0].begin
321
322    @property
323    def end(self) -> float:
324        """
325        End timestamp.
326
327        Corresponds to a TextGrid Interval's xmin.
328
329        This is a property and the actual value is generated from the last
330        PatInterval of this PatTier.
331        """
332        # This is slightly counterintuitive, but the last interval is in fact
333        # empty and only represents the final boundary. So its `begin` is the
334        # final boundary.
335        return self[-1].begin
336
337    @property
338    def is_point_tier(self) -> bool:
339        """Is this Tier a PointTier."""
340        return False
341
342    def boundary_at_time(
343            self, time: float, epsilon: float) -> PatInterval | None:
344        """
345        If there is a boundary at time, return it.
346
347        Returns None, if there is no boundary at time.
348
349        'Being at time' is defined as being within patkit epsilon of the given
350        timestamp.
351        """
352        for interval in self:
353            if interval.is_at_time(time=time, epsilon=epsilon):
354                return interval
355        return None
356
357    def get_interval_by_category(
358        self,
359        interval_category: IntervalCategory,
360        label: str | None = None
361    ) -> PatInterval:
362        """
363        Return the Interval matching the category in this Tier.
364
365        If interval_category is FIRST_LABELED or LAST_LABELED, the label should
366        be specified as well.
367
368        Parameters
369        ----------
370        interval_category : IntervalCategory
371            The category to search for.
372        label : Optional[str], optional
373            Label to search for when doing a label based category search, by
374            default None
375
376        Returns
377        -------
378        PatInterval
379            The matching PatInterval
380        """
381        if interval_category is IntervalCategory.FIRST_NON_EMPTY:
382            for interval in self:
383                if interval.label:
384                    return interval
385
386        if interval_category is IntervalCategory.LAST_NON_EMPTY:
387            for interval in reversed(self):
388                if interval.label:
389                    return interval
390
391        if interval_category is IntervalCategory.FIRST_LABELED:
392            for interval in self:
393                if interval.label == label:
394                    return interval
395
396        if interval_category is IntervalCategory.LAST_LABELED:
397            for interval in reversed(self):
398                if interval.label == label:
399                    return interval
400
401    def get_labels(
402            self, time_vector: np.ndarray, epsilon: float | None = None,
403    ) -> np.ndarray:
404        """
405        Get the labels at the times in the `time_vector`.
406
407        Parameters
408        ----------
409        time_vector : np.ndarray
410            Time stamps to retrieve the labels for.
411        epsilon : float | None
412            The precision (in seconds) to use in comparisons. The default value
413            None will result in PATKIT_EPSILON, being used. For expected
414            behaviour for `PointTiers`, `configuration.data_config.epsilon`
415            should be passed here.
416
417        Returns
418        -------
419        np.ndarray
420            This array contains the labels as little endian Unicode strings.
421        """
422        max_label = max(
423            [len(element.label) for element in self
424             if element.label is not None]
425        )
426        labels = np.empty(len(time_vector), dtype=f"<U{max_label}")
427        for (i, time) in enumerate(time_vector):
428            labels[i] = self.label_at(time, epsilon)
429        return labels
430
431    def label_at(self, time: float, epsilon: float | None = None) -> str:
432        """
433        Get the label at the given time.
434
435        Parameters
436        ----------
437        time : float
438            Time in seconds to retrieve the label for.
439        epsilon : float | None
440            The precision (in seconds) to use in comparisons. The default value
441            None will result in PATKIT_EPSILON, being used. For expected
442            behaviour for `PointTiers`, `configuration.data_config.epsilon`
443            should be passed here.
444
445        Returns
446        -------
447        str
448            The label string.
449        """
450        if time < self.begin or time > self.end:
451            return ""
452
453        for element in self:
454            if element.contains(time=time, epsilon=epsilon):
455                return element.label
456
457    def intersects(self, xlim: list[float, float]) -> list[PatAnnotation]:
458        """
459        List the items that intersect with the given time interval.
460
461        Parameters
462        ----------
463        xlim : list[float, float]
464            The time interval in seconds.
465
466        Returns
467        -------
468        list[PatAnnotation]
469            List of PatAnnotations that intersect with the interval.
470        """
471        in_limits = []
472        if len(self) > 0:
473            if isinstance(self[0], PatInterval):
474                for item in self:
475                    if item.end is None:
476                        if xlim[1] < item.begin:
477                            continue
478                        if item.begin < xlim[0]:
479                            continue
480                    elif item.end < xlim[0] or xlim[1] < item.begin:
481                        continue
482                    in_limits.append(item)
483            else:
484                for item in self:
485                    if item.time < xlim[1] and xlim[0] < item.time:
486                        in_limits.append(item)
487
488        return in_limits
489
490    def scramble(self):
491        boundaries = np.linspace(
492            start=self.begin,
493            stop=self.end,
494            num=len(self)
495        )
496        for item, boundary in zip(self, boundaries):
497            item.time = boundary

500class PatGrid(OrderedDict):
501    """
502    TextGrid representation which makes editing easier.
503
504    PatGrid is a OrderedDict very similar to Python textgrids TextGrid, but
505    made up of PatTiers that in turn contain intervals or points as doubly
506    linked lists instead of just lists. See the relevant classes for more
507    details.
508    """
509
510    def __init__(self, textgrid: TextGrid | Path | str) -> None:
511        """
512        Initialize a PatGrid.
513
514        Parameters
515        ----------
516        textgrid : TextGrid | Path | str
517            An already parsed textgrids.TextGrid object, or a path to a
518            TextGrid file on disk.
519        """
520        super().__init__()
521
522        if isinstance(textgrid, (Path, str)):
523            textgrid = TextGrid(textgrid)
524
525        for tier_name in textgrid:
526            self[tier_name] = PatTier.from_textgrid_tier(textgrid[tier_name])
527
528    # def as_textgrid(self):
529    #     pass
530
531    def __deepcopy__(self, memo):
532        cls = self.__class__
533        new_copy = cls.__new__(cls)
534        memo[id(self)] = new_copy
535        for tier_name, tier in self.items():
536            new_copy[tier_name] = deepcopy(tier)
537        return new_copy
538
539    def __repr__(self):
540        representation = "PatGrid:"
541        for tier_name in self:
542            representation += f"\n\tTier: {tier_name}"
543            for item in self[tier_name]:
544                if isinstance(item, PatInterval):
545                    representation += f"\n\t\t{item}"
546        return representation
547
548    @property
549    def begin(self) -> float:
550        """
551        Begin timestamp.
552
553        Corresponds to a TextGrids xmin.
554
555        This is a property and the actual value is generated from the first
556        PatTier of this PatGrid.
557        """
558        key = list(self.keys())[0]
559        return self[key].begin
560
561    @property
562    def end(self) -> float:
563        """
564        End timestamp.
565
566        Corresponds to a TextGrids xmax.
567
568        This is a property and the actual value is generated from the first
569        PatTier of this PatGrid.
570        """
571        # First Tier
572        key = list(self.keys())[0]
573        # Return the end of the first Tier.
574        return self[key].end
575
576    def format_long(self) -> str:
577        """Format self as long format TextGrid."""
578        out = long_header.format(self.begin, self.end, len(self))
579        tier_count = 1
580        for name, tier in self.items():
581            if tier.is_point_tier:
582                tier_type = 'PointTier'
583                element_type = 'points'
584            else:
585                tier_type = 'IntervalTier'
586                element_type = 'intervals'
587            out += long_tier.format(tier_count,
588                                    tier_type,
589                                    name,
590                                    self.begin,
591                                    self.end,
592                                    element_type,
593                                    len(tier)-1)
594            for element_count, element in enumerate(tier, 1):
595                if tier.is_point_tier:
596                    out += long_point.format(element_count,
597                                             element.time,
598                                             element.label)
599                elif element.next:
600                    out += long_interval.format(element_count,
601                                                element.begin,
602                                                element.end,
603                                                element.label)
604                else:
605                    # The last interval does not contain anything. It only
606                    # marks the end of the file and final interval's end. That
607                    # info got already used by element.end (which is
608                    # element.next.begin) above.
609                    pass
610        return out
611
612    def get_labels(
613            self, time_vector: np.ndarray, epsilon: float | None = None
614    ) -> dict[str, np.ndarray]:
615        """
616        Get the labels at the times in the `time_vector`.
617
618        Parameters
619        ----------
620        time_vector : np.ndarray
621            Time values to get the labels for.
622        epsilon : float | None
623            The precision (in seconds) to use in comparisons. The default value
624            None will result in PATKIT_EPSILON, being used. For expected
625            behaviour for `PointTiers`, `configuration.data_config.epsilon`
626            should be passed here.
627
628        Returns
629        -------
630        dict[str, np.ndarray]
631            Dictionary of the labels indexed by tier name.
632        """
633        labels = {}
634        for tier_name in self:
635            labels[tier_name] = self[tier_name].get_labels(
636                time_vector, epsilon=epsilon)
637
638        return labels
