217    def setup_axes(self) -> None:
218        """
219        Clear the figure and set up the required subplots.
220        """
221        self.figure.clear()
222        self.data_axes = []
223        self.tier_axes = []
224
225        height_ratios = [
226            self.gui_config.data_and_tier_height_ratios.data_axes,
227            self.gui_config.data_and_tier_height_ratios.tier_axes
228        ]
229        self.main_grid_spec = self.figure.add_gridspec(
230            nrows=2,
231            ncols=1,
232            hspace=0,
233            wspace=0,
234            height_ratios=height_ratios,
235        )
236
237        number_of_data_axes = self.gui_config.number_of_data_axes
238        self.data_grid_spec = self.main_grid_spec[0].subgridspec(
239            number_of_data_axes, 1, hspace=0, wspace=0)
240
241        data_axes_params = None
242        if self.gui_config.general_axes_params:
243            if self.gui_config.general_axes_params is not None:
244                data_axes_params = self.gui_config.general_axes_params
245
246        for i, axes_name in enumerate(self.gui_config.data_axes):
247            sharex = False
248            if self.gui_config.data_axes[axes_name].sharex:
249                sharex = self.gui_config.data_axes[axes_name].sharex
250            elif (data_axes_params is not None and
251                  data_axes_params.sharex is not None):
252                sharex = data_axes_params.sharex
253
254            if i != 0 and sharex:
255                ax = self.figure.add_subplot(
256                    self.data_grid_spec[i],
257                    sharex=self.data_axes[0])
258            else:
259                ax = self.figure.add_subplot(
260                    self.data_grid_spec[i])
261            self.data_axes.append(ax)
262
263        self.canvas.draw_idle()
264
265    def clear_axes(self) -> None:
266        """Clear all data axes."""
267        for axes in self.data_axes + self.tier_axes:
268            axes.cla()

270    def draw_plots(
271        self,
272        recording: Recording,
273        patgrid: list,
274        xlim: tuple[float, float],
275        mode: AnnotatorMode,
276        exercise_mode: ExerciseMode,
277        title: str,
278    ) -> None:
279        """
280        Dynamically calculate grid specs and draw the data and tiers.
281        """
282        if recording.excluded:
283            self.display_exclusion()
284
285        self.setup_axes()
286
287        if 'MonoAudio' in recording.modalities:
288            self.data_axes[0].set_title(title)
289        else:
290            self.data_axes[0].set_title(
291                title + "\nNOTE: Audio missing.")
292            return
293
294        for axes in self.tier_axes:
295            axes.remove()
296        self.tier_axes = []
297        # if current_recording.patgrid:
298        if patgrid:
299            nro_tiers = len(patgrid)
300            self.tier_grid_spec = self.main_grid_spec[1].subgridspec(
301                nro_tiers, 1, hspace=0, wspace=0)
302            for axes_counter, tier in enumerate(patgrid):
303                axes = self.figure.add_subplot(
304                    self.tier_grid_spec[axes_counter],
305                    sharex=self.data_axes[0])
306                axes.set_yticks([])
307                self.tier_axes.append(axes)
308
309        # Hide all bottom ticks for data axes initially
310        for axes in self.data_axes:
311            axes.xaxis.set_tick_params(bottom=False, labelbottom=False)
312        self.data_axes[0].xaxis.set_tick_params(top=True, labeltop=True)
313
314        # Hide bottom ticks for all tier axes EXCEPT the last one
315        for axes in self.tier_axes[:-1]:
316            axes.xaxis.set_tick_params(bottom=False, labelbottom=False)
317
318        # Explicitly enforce ticks ON for the bottom-most axis so MultiCursor
319        # doesn't flush them away during updates.
320        if self.tier_axes:
321            self.tier_axes[-1].xaxis.set_tick_params(
322                bottom=True, labelbottom=True)
323        elif self.data_axes:
324            self.data_axes[-1].xaxis.set_tick_params(
325                bottom=True, labelbottom=True)
326
327        if 'MonoAudio' not in recording.modalities:
328            return
329
330        audio = recording.modalities['MonoAudio']
331        if audio.go_signal is None:
332            stimulus_onset = 0
333        else:
334            stimulus_onset = audio.go_signal
335
336        wav = audio.data
337        wav_time = audio.timevector - stimulus_onset
338
339        if self.gui_config.xlim is not None:
340            xlim = self.gui_config.xlim
341        elif self.gui_config.auto_xlim:
342            x_minimums = []
343            x_maximums = []
344            modalities_to_check = self.gui_config.plotted_modality_names()
345            modalities_to_check.add("MonoAudio")
346            for name in modalities_to_check:
347                if name in recording:
348                    x_minimums.append(
349                        recording[name].timevector[0] - stimulus_onset
350                    )
351                    x_maximums.append(
352                        recording[name].timevector[-1] - stimulus_onset
353                    )
354            xlim = (np.min(x_minimums)-.05, np.max(x_maximums)+.05)
355
356        axes_counter = 0
357        for axes_name in self.gui_config.data_axes:
358            self.data_axes[axes_counter].grid(False)
359            match axes_name:
360                case "spectrogram":
361                    if self.gui_config.data_axes[axes_name].ylim is not None:
362                        ylim = self.gui_config.data_axes[axes_name].ylim
363                    else:
364                        ylim = (0, 10500)
365                    plot_spectrogram(self.data_axes[axes_counter],
366                                     waveform=wav,
367                                     ylim=ylim,
368                                     sampling_frequency=audio.sampling_rate,
369                                     extent_on_x=(wav_time[0], wav_time[-1]))
370                case "spectrogram2":
371                    if self.gui_config.data_axes[axes_name].ylim is not None:
372                        ylim = self.gui_config.data_axes[axes_name].ylim
373                    else:
374                        ylim = (0, 10500)
375                    plot_spectrogram2(
376                        self.data_axes[axes_counter],
377                        waveform=wav,
378                        ylim=ylim,
379                        sampling_frequency=audio.sampling_rate,
380                        extent_on_x=(wav_time[0], wav_time[-1]),
381                        mode=self.gui_config.color_scheme
382                    )
383                case "wav":
384                    plot_wav(
385                        ax=self.data_axes[axes_counter],
386                        waveform=wav,
387                        wav_time=wav_time,
388                        xlim=xlim,
389                        mode=self.gui_config.color_scheme
390                    )
391                case _:
392                    if not recording.excluded:
393                        self.plot_modality_axes(
394                            recording=recording,
395                            xlim=xlim,
396                            axes_number=axes_counter,
397                            axes_name=axes_name,
398                            zero_offset=stimulus_onset,
399                            ylim=self.gui_config.data_axes[axes_name].ylim,
400                        )
401            axes_counter += 1
402
403        self.animators = []
404        iterator = zip(patgrid.items(),
405                       self.tier_axes, strict=True)
406        for (name, tier), axis in iterator:
407            axis.grid(False)
408            boundaries_by_axis = []
409
410            boundary_set, _ = plot_patgrid_tier(
411                axes=axis,
412                tier=tier,
413                time_offset=stimulus_onset,
414                text_y=.5,
415                xlim=xlim,
416            )
417            boundaries_by_axis.append(boundary_set)
418            axis.set_ylabel(
419                name, rotation=90, horizontalalignment="center",
420                verticalalignment="center")
421            axis.set_xlim(xlim)
422            if name in self.gui_config.pervasive_tiers:
423                for data_axis in self.data_axes:
424                    boundary_set = plot_patgrid_tier(
425                        axes=data_axis,
426                        tier=tier,
427                        time_offset=stimulus_onset,
428                        draw_text=False)[0]
429                    boundaries_by_axis.append(boundary_set)
430
431            # Change rows to be individual boundaries instead of axis. This
432            # makes it possible to create animators for each boundary as
433            # represented by multiple lines on different axes.
434            boundaries_by_boundary = list(map(list, zip(*boundaries_by_axis)))
435
436            tier_limits = [
437                xlim[0] + stimulus_onset, xlim[1] + stimulus_onset]
438            tier_in_limits = tier.intersects(xlim=tier_limits)
439            if (
440                mode is AnnotatorMode.ANALYSE or
441                exercise_mode is ExerciseMode.ANSWER
442            ):
443                for boundaries, interval in zip(
444                        boundaries_by_boundary, tier_in_limits, strict=True):
445                    animator = BoundaryAnimator(
446                        main_window=self.main_window,
447                        boundaries=boundaries,
448                        segment=interval,
449                        epsilon=self.data_config.epsilon,
450                        time_offset=stimulus_onset)
451                    animator.connect()
452                    self.animators.append(animator)
453        if self.tier_axes:
454            self.tier_axes[-1].set_xlabel("Time (s)")
455
456        # Save clean ticks to avoid infinite accumulation
457        # when adding selection ticks
458        self.original_xticks = self.data_axes[0].get_xticks()
459        self.original_yticks = {ax: ax.get_yticks() for ax in self.data_axes}
460
461        self.update_selection_cursors(recording)
462
463        self.playback_cursor_lines = []
464        for ax in self.data_axes + self.tier_axes:
465            line = ax.axvline(
466                x=0,
467                color=DefaultCursorColors.PLAYBACK,
468                linestyle='-',
469                visible=False
470            )
471            self.playback_cursor_lines.append(line)
472
473        # Restore MultiCursor
474        self.update_multicursor()
475
476        # Align the y-labels nicely across all subplots
477        self.figure.align_ylabels()

479    def update_selection_cursors(self, recording: Recording) -> None:
480        """
481        Update the selection cursors across the subplots.
482
483        Fast path for rendering the selection cursors and tick
484        modifications without a full canvas wipe/rebuild.
485        """
486        # Clean up old drawn selection lines
487        for artist in getattr(self, 'selection_artists', []):
488            try:
489                artist.remove()
490            except Exception:
491                pass
492        self.selection_artists = []
493
494        # Reset ticks to their clean original states
495        if hasattr(self, 'original_xticks') and len(self.original_xticks) > 0:
496            self.data_axes[0].set_xticks(self.original_xticks)
497
498            # Remove deepskyblue from regular ticks
499            for label in self.data_axes[0].get_xticklabels():
500                label.set_color("black")
501            if self.tier_axes:
502                for label in self.tier_axes[-1].get_xticklabels():
503                    label.set_color("black")
504
505        if hasattr(self, 'original_yticks'):
506            for ax, yticks in self.original_yticks.items():
507                ax.set_yticks(yticks)
508
509        selected_time = recording.annotations['selected_time']
510        selected_freq = recording.annotations['selected_frequency']
511
512        # Apply the new selection
513        if selected_time > -1:
514            old_ticks = self.original_xticks
515            if len(old_ticks) > 2:
516                self.data_axes[0].set_xticks(
517                    [old_ticks[1], selected_time, old_ticks[-2]]
518                )
519            elif len(old_ticks) > 0:
520                self.data_axes[0].set_xticks(
521                    [old_ticks[0], selected_time, old_ticks[-1]]
522                )
523
524            xtick_labels = self.data_axes[0].get_xticklabels()
525            if len(xtick_labels) > 1:
526                xtick_labels[1].set_color(color=DefaultCursorColors.SELECTION)
527
528            if self.tier_axes:
529                xtick_labels = self.tier_axes[-1].get_xticklabels()
530                if len(xtick_labels) > 1:
531                    xtick_labels[1].set_color(
532                        color=DefaultCursorColors.SELECTION)
533
534            for axes in self.data_axes:
535                current_ylim = axes.get_ylim()
536                vline = axes.axvline(
537                    x=selected_time,
538                    linestyle=':',
539                    color=DefaultCursorColors.SELECTION,
540                    lw=1
541                )
542                self.selection_artists.append(vline)
543
544                lines = axes.get_lines()
545                colors = []
546                yticks = axes.get_yticks()
547                for line in lines:
548                    if line in self.playback_cursor_lines:
549                        continue
550                    if line in self.selection_artists:
551                        continue
552                    if len(line.get_xdata()) <= 2:
553                        continue
554
555                    xdata = line.get_xdata()
556                    if len(xdata) == 0:
557                        continue
558
559                    index = np.argmin(np.abs(xdata - selected_time))
560                    y_value = line.get_ydata()[index]
561                    color = line.get_color()
562
563                    hline = axes.axhline(
564                        y=y_value, linestyle=':', color=color, lw=1)
565                    self.selection_artists.append(hline)
566                    yticks = np.append(yticks, y_value)
567                    colors.append(color)
568
569                axes.set_yticks(yticks)
570
571                labels = axes.get_yticklabels()
572                ytick_lines = axes.yaxis.get_ticklines()
573                for i, color in enumerate(colors):
574                    if len(ytick_lines) > i*2+5:
575                        ytick_lines[i*2+4].set_color(color)
576                        ytick_lines[i*2+5].set_color(color)
577                    if len(labels) > i+2:
578                        labels[i+2].set_color(color)
579                axes.set_ylim(current_ylim)
580        else:
581            old_ticks = self.original_xticks
582            if len(old_ticks) > 2:
583                self.data_axes[0].set_xticks([old_ticks[1], old_ticks[-2]])
584
585        if selected_freq > -1:
586            for i, name in enumerate(self.gui_config.data_axes):
587                if "spectrogram" in name:
588                    axes = self.data_axes[i]
589                    yticks = axes.get_yticks()
590                    axes.set_yticks(np.append(yticks, selected_freq))
591
592                    hline = axes.axhline(
593                        y=selected_freq,
594                        linestyle=':',
595                        color=DefaultCursorColors.SELECTION,
596                        lw=1
597                    )
598                    self.selection_artists.append(hline)
599
600                    labels = axes.get_yticklabels()
601                    if len(labels) > 2:
602                        labels[2].set_color(
603                            color=DefaultCursorColors.SELECTION)
604                    ytick_lines = axes.yaxis.get_ticklines()
605                    if len(ytick_lines) > 5:
606                        ytick_lines[4].set_color(
607                            color=DefaultCursorColors.SELECTION)
608                        ytick_lines[5].set_color(
609                            color=DefaultCursorColors.SELECTION)
610
611            for axes in self.tier_axes:
612                vline = axes.axvline(
613                    x=selected_time,
614                    linestyle=':',
615                    color=DefaultCursorColors.SELECTION,
616                    lw=1
617                )
618                self.selection_artists.append(vline)
619
620        # Force a synchronous draw to lock in the pixels immediately
621        # and clear the MultiCursor's background cache.
622        self.canvas.draw()
623        if self.multicursor is not None:
624            self.multicursor.background = None

626    def plot_modality_axes(
627        self,
628        axes_number: int,
629        axes_name: str,
630        recording: Recording,
631        xlim: tuple[float, float],
632        zero_offset: float = 0,
633        ylim: list[float, float] | None = None,
634    ) -> None:
635        """
636        Plot modalities on a data_axes.
637        """
638        axes_params = self.gui_config.data_axes[axes_name]
639        data_axes_params = self.gui_config.general_axes_params.data_axes
640        plot_modality_names = axes_params.modalities
641
642        if ylim is None:
643            if data_axes_params is None and axes_params is None:
644                ylim = None  # (-0.075, 1.075)
645            elif data_axes_params.ylim is None and axes_params.ylim is None:
646                if (
647                    not data_axes_params.auto_ylim and
648                    not axes_params.auto_ylim
649                ):
650                    ylim = None  # (-0.075, 1.075)
651                else:
