  66class Answer(AbstractDataContainer, UserList):
  67    """
  68    Answer is an answer to an Exercise and consists of PatGrids.
  69
  70    The PatGrids correspond to the Recordings that the Exercise is based on.
  71    """
  72
  73    def __init__(
  74        self,
  75        container: Exercise,
  76        scenario: Session,
  77        metadata: AnswerMetadata,
  78        name: str,
  79        cursor: int = 0,
  80        file_info: FileInformation | None = None,
  81    ):
  82        """
  83        Initialize an Answer object containing PatGrids for an Exercise.
  84
  85        Parameters
  86        ----------
  87        container : Exercise
  88            The Exercise this Answer belongs to.
  89        scenario : Session
  90            The Session containing the recordings.
  91        name : str
  92            The name of the answer.
  93        cursor : int
  94            The cursor index pointing to the active recording.
  95        file_info : FileInformation | None
  96            The FileInformation metadata for the answer.
  97        """
  98        if file_info is None:
  99            file_info = FileInformation()
 100
 101        super().__init__(name=name, metadata=metadata,
 102                         container=container, file_info=file_info)
 103        self.container = container
 104        self.scenario = scenario
 105        self.cursor = cursor
 106
 107        for recording in self.scenario:
 108            patgrid = deepcopy(recording.patgrid)
 109            self.append(patgrid)
 110
 111        # TODO: this will probably break loading edited answers
 112        if self.metadata.scramble:
 113            for patgrid in self:
 114                for tier in patgrid:
 115                    patgrid[tier].scramble()
 116            self.edited = [False for i in range(len(self))]
 117        else:
 118            self.edited = [True for i in range(len(self))]
 119
 120    def __repr__(self) -> str:
 121        representation = f"Answer - name: {self.name}, author: {self.author}"
 122        for patgrid in self:
 123            patgrid_string = f"{patgrid}"
 124            representation += indent(text=patgrid_string, prefix="\t")
 125        return representation
 126
 127    @property
 128    def patkit_data_path(self) -> Path:
 129        """Dynamically compute the path to the answer directory."""
 130        # Route example answers directly to the example directory
 131        if (
 132            self.container.example is not None and
 133            self is self.container.example
 134        ):
 135            return self.container.patkit_path / PatkitDirectory.EXAMPLE
 136
 137        return (
 138            self.container.patkit_path / PatkitDirectory.ANSWERS / self.name
 139        )
 140
 141    def current(self) -> PatGrid:
 142        """
 143        The PatGrid at the current index.
 144
 145        Returns
 146        -------
 147        PatGrid
 148            The PatGrid.
 149        """
 150        return self[self.cursor]
 151
 152    def go_to_recording(self, index: int) -> int:
 153        """
 154        Go to Recording at index.
 155
 156        Returns
 157        -------
 158        int
 159            index of current recording
 160        """
 161        if len(self) > index and index >= 0:
 162            self.cursor = index
 163        return self.cursor
 164
 165    def next(self) -> int:
 166        """
 167        Go to next Recording.
 168
 169        Returns
 170        -------
 171        int
 172            index of next recording
 173        """
 174        if len(self) > self.cursor + 1:
 175            self.cursor += 1
 176        return self.cursor
 177
 178    def previous(self) -> int:
 179        """
 180        Go to previous Recording.
 181
 182        Returns
 183        -------
 184        int
 185            index of previous recording
 186        """
 187        if self.cursor > 0:
 188            self.cursor -= 1
 189        return self.cursor

 192class Exercise(AbstractDataContainer, UserDict):
 193    """
 194    Exercise is list of Answers relating to a Scenario (Session).
 195
 196    If no example Answers are provided, the current PatGrids in the Session are
 197    copied and used as the example.
 198    """
 199
 200    def __init__(
 201        self,
 202        scenario: Session,
 203        name: str,
 204        metadata: ExerciseMetadata,
 205        answers: list[Answer] | None = None,
 206        example: dict[str, Answer] | None = None,
 207        index: int = 0,
 208        file_info: FileInformation | None = None,
 209    ):
 210        """
 211        Initialize an Exercise.
 212
 213        Parameters
 214        ----------
 215        scenario : Session
 216            The Session containing the recordings for this exercise.
 217        answers : list[Answer] | None
 218            A list of Answer objects to populate the exercise.
 219        example : dict[str, Answer] | None
 220            The example scrambled Answer to present to the user.
 221        index : int
 222            The cursor index.
 223        """
 224        super().__init__(name=name, metadata=metadata, file_info=file_info)
 225        if example is None:
 226            _logger.debug("Creating an example answer")
 227            metadata = AnswerMetadata(
 228                scramble=False,
 229            )
 230            self.example = Answer(
 231                container=self,
 232                scenario=scenario,
 233                metadata=metadata,
 234                name=PatkitDirectory.EXAMPLE,
 235            )
 236        else:
 237            self.example = example
 238        self.scenario = scenario
 239
 240        if answers is not None:
 241            for answer in answers:
 242                self[answer.name] = answer
 243
 244        self.cursor = index
 245
 246    def __repr__(self) -> str:
 247        representation = "Exercise:\n"
 248        scenario_repr = f"{self.scenario}"
 249        scenario_repr = indent(text=scenario_repr, prefix="\t\t")
 250        representation += f"\tScenario:\n{scenario_repr}\n"
 251        example_repr = f"{self.example}"
 252        example_repr = indent(text=example_repr, prefix="\t")
 253        representation += f"\tExample:\n{example_repr}\n"
 254        representation += "\tAnswers:\n"
 255        answers_string = ""
 256        for name in list(self.keys()):
 257            answers_string += f"{name}:\n{self[name]}"
 258        representation += indent(text=answers_string, prefix="\t\t")
 259        return representation
 260
 261    @property
 262    def current_answer(self) -> Answer:
 263        return self[list(self.keys())[self.cursor]]
 264
 265    @property
 266    def patkit_path(self) -> Path | None:
 267        return self.file_info.patkit_path
 268
 269    @patkit_path.setter
 270    def patkit_path(self, path: Path) -> None:
 271        self.file_info.patkit_path = path
 272
 273    def new_blank_answer(
 274        self,
 275        cursor: int = 0,
 276        name: str | None = None,
 277        author: str = "",
 278    ) -> None:
 279        """
 280        Create a new blank Answer and append it to this Exercise.
 281
 282        Parameters
 283        ----------
 284        cursor : int
 285            Cursor position for the new Answer.
 286        name : str | None
 287            Name of the new Answer, by default None. If None is passed the new
 288            name will be autogenerated and have the form
 289            `f"Answer_{len(self)+1}"`.
 290        author : str
 291            Name of the new Answer's author, by default an empty string.
 292        """
 293        _logger.debug("Creating a blank answer.")
 294
 295        # TODO 1.0: Is this the best place to name answers?
 296        if name is None:
 297            name = f"Answer_{len(self)+1}"
 298
 299        metadata = AnswerMetadata(
 300            scramble=True,
 301            author=author
 302        )
 303        blank = Answer(
 304            container=self,
 305            scenario=self.scenario,
 306            name=name,
 307            metadata=metadata,
 308            cursor=cursor,
 309        )
 310        self[name] = blank

 313class Manifest(UserList):
 314    """
 315    Manifest is a list of Scenario paths as strings.
 316    """
 317    @staticmethod
 318    def read_manifest(path: Path) -> list[str] | None:
 319        """
 320        Static method for reading a Manifest file contents.
 321
 322        Parameters
 323        ----------
 324        path : Path
 325            Path of the file.
 326
 327        Returns
 328        -------
 329        list[str] | None
 330            Either the contents of the file or None if the file did not exist.
 331        """
 332        if not path.exists():
 333            return None
 334        with open(path, 'r', encoding='UTF-8') as file:
 335            lines = []
 336            for line in file:
 337                line = line.rstrip()
 338                if line == "":
 339                    continue
 340                if line == "None":
 341                    continue
 342                lines.append(line)
 343            return lines
 344        return None
 345
 346    def __init__(self, path: Path | None = None):
 347        super().__init__()
 348        if path is not None:
 349            path = path.resolve()
 350            self.path = path
 351            scenario_paths = Manifest.read_manifest(path)
 352            if scenario_paths is not None:
 353                self.extend(scenario_paths)
 354
 355    @property
 356    def scenarios(self) -> list[str]:
 357        """
 358        The contents of this Manifest as a list.
 359
 360        Returns
 361        -------
 362        list[str]
 363            The contents.
 364        """
 365        return list(self)
 366
 367    def save(self):
 368        """
 369        Write this manifest into its file.
 370        """
 371        if self.path is not None:
 372            with open(self.path, 'w', encoding='UTF-8') as file:
 373                for scenario_path in self:
 374                    file.write(f"{scenario_path}\n")
 375
 376    def to_string(self):
 377        string = "Manifest:\n"
 378        string += f"\tpath: {self.path}\n"
 379        for item in self:
 380            string += f"\tcontent: {item}\n"
 381        return string
 382
 383    def append(self, new_item: str):
 384        if new_item is None:
 385            return
 386        if not isinstance(new_item, str):
 387            new_item = str(new_item)
 388        if new_item not in self:
 389            super().append(new_item)

 392class Session(AbstractDataContainer, UserList):
 393    """
 394    The metadata and Recordings of a recording session.
 395
 396    This class behaves exactly like a list of Recordings with some extra
 397    fields. While some legacy code may be left behind, the preferred idiom for
 398    iterating over the recordings is `for recording in recording_session:` or
 399    `for recording in tqdm(recording_session):` if you want a progress bar on
 400    the command line.
 401
 402    Sessions can also hold aggregate data in the form of Statistics.
 403    """
 404
 405    def __init__(
 406            self,
 407            name: str,
 408            config: SessionConfig,
 409            file_info: FileInformation,
 410            recordings: list[Recording] | None = None,
 411            statistics: dict[str, Statistic] | None = None,
 412            exercise: Exercise | None = None,
 413    ) -> None:
 414        super().__init__(
 415            container=None,
 416            name=name,
 417            metadata=config,
 418            file_info=file_info,
 419            statistics=statistics)
 420
 421        if recordings is not None:
 422            for recording in recordings:
 423                if not recording.container:
 424                    recording.container = self
 425            self.extend(recordings)
 426
 427        self.config = config
 428        self.exercise = exercise
 429
 430    def __repr__(self) -> str:
 431        representation = "Session:\n"
 432        for recording in self:
 433            representation += f"\tRecording: {recording.basename},\n"
 434        return representation
 435
 436    @property
 437    def recordings(self) -> list[Recording]:
 438        """
 439        Property to access the list of Recordings directly.
 440
 441        Returns
 442        -------
 443        list[Recording]
 444            The list of this Session's Recordings.
 445        """
 446        return self.data

 449class Recording(AbstractDataContainer, UserDict):
 450    """
 451    A Recording is a dictionary of 0-n synchronised Modalities.
 452
 453    Adding modalities can be done by `recording[name] = modality`, but
 454    `recording.add_modality` is preferred as a safer way which checks for
 455    overwriting. The reason Recording is a dictionary is to make it possible to
 456    iterate with the idiom `for modality_name in recording`.
 457
 458    The recording also contains the non-modality-specific metadata
 459    (participant, speech content, etc.) as a dictionary, as well as the
 460    textgrid for the whole recording.
 461
 462    In general, inheriting should not be necessary, but if it is, inheriting
 463    classes should call `self._read_textgrid()` after calling
 464    `super.__init__()` (with correct arguments) and doing any updates to
 465    `self.meta['textgrid']` that are necessary.
 466    """
 467    container: Session
 468
 469    def __init__(
 470            self,
 471            metadata: RecordingMetaData,
 472            file_info: FileInformation,
 473            container: Session | None = None,
 474            excluded: bool = False,
 475    ) -> None:
 476        """
 477        Construct a mainly empty recording without modalities.
 478
 479        Modalities and annotations get added after constructions with their own
 480        add_[modality or annotation] functions.
 481
 482        NOTE: `after_modalities_init` should be called on each new Recording
 483        after modalities have been loaded. It ensures that there is at least a
 484        minimal TextGrid in place to facilitate GUI functions.
 485
 486        Parameters
 487        ----------
 488        metadata : RecordingMetaData
 489            Some of the contents of the meta data are available as properties.
 490        excluded : bool, optional
 491            _description_, by default False
 492        """
 493        if file_info.recorded_meta_file:
 494            name = file_info.recorded_meta_file
 495        else:
 496            name = file_info.patkit_meta_file
 497        super().__init__(
 498            container=container, name=name,
 499            metadata=metadata, file_info=file_info)
 500
 501        self.excluded = excluded
 502        self.textgrid_path = None
 503        self.textgrid = None
 504        self.patgrid = None
 505        self.annotations = {}
 506
 507    @property
 508    def modalities(self) -> dict[str, Modality]:
 509        """
 510        Dictionary of the modalities of this Recording.
 511
 512        Returns
 513        -------
 514        dict[str, Modality]
 515            The dictionary of the Modalities.
 516        """
 517        return self.data
 518
 519    @property
 520    def basename(self) -> str:
 521        """Filename of this Recording without extensions."""
 522        return self.file_info.basename
 523
 524    def _read_textgrid(self) -> textgrids.TextGrid | None:
 525        """
 526        Read the textgrid specified in self.meta.
 527
 528        If file does not exist or reading fails, recovery is attempted
 529        by logging an error and creating an empty textgrid for this
 530        recording.
 531        """
 532        textgrid = None
 533        if self.textgrid_path.is_file():
 534            try:
 535                textgrid = textgrids.TextGrid(self.textgrid_path)
 536                _logger.info("Read textgrid in %s.",
 537                             self.textgrid_path)
 538            except Exception as exception:
 539                _logger.critical(
 540                    "Could not read textgrid in %s.", self.textgrid_path)
 541                _logger.critical(
 542                    "Failed with: %s.", str(exception))
 543        else:
 544            notice = 'Note: ' + str(self.textgrid_path) + " did not exist."
 545            _logger.warning(notice)
 546        return textgrid
 547
 548    def identifier(self) -> str:
 549        """
 550        Generate a unique identifier for this Recording from metadata.
 551
 552        Returns
 553        -------
 554        str
 555            prompt followed by time of recording.
 556        """
 557        return (f"{self.metadata.prompt} "
 558                f"{str(self.metadata.time_of_recording)}")
 559
 560    def exclude(self) -> None:
 561        """
 562        Set `self.excluded` to True with a method.
 563
 564        This method exists to facilitate list comprehensions being used
 565        for excluding recordings e.g.
 566        [recording.exclude() for recording in recordings if in some_list].
 567        """
 568        self.excluded = True
 569
 570    def write_textgrid(self, filepath: Path = None) -> None:
 571        """
 572        Save this recording's textgrid to file.
 573
 574        Keyword argument:
 575        filepath -- string specifying the path and name of the
 576            file to be written. If filepath is not specified, this
 577            method will try to overwrite the textgrid specified in
 578            self.meta.
 579
 580            If filepath is specified, subsequent calls to this
 581            function will write into the new path rather than
 582            the original one.
 583        """
 584        try:
 585            if filepath:
 586                self.textgrid_path = filepath
 587                self.textgrid.write(filepath)
 588            else:
 589                self.textgrid.write(self.textgrid_path)
 590        except Exception as exception:
 591            _logger.critical(
 592                "Could not write textgrid to %s.", str(self.textgrid_path))
 593            _logger.critical(
 594                "TextGrid save failed with error: %s", str(exception))
 595
 596    def add_modality(
 597            self, modality: Modality, replace: bool = False
 598    ) -> None:
 599        """
 600        This method adds a new Modality object to the Recording.
 601
 602        Replacing a modality has to be specified otherwise if a
 603        Modality with the same name already exists in this Recording
 604        and the `replace` argument is not True, an Error is raised.
 605
 606        Arguments:
 607        modality -- object of type Modality to be added to
 608            this Recording.
 609
 610        Keyword arguments:
 611        replace -- a boolean indicating if an existing Modality should
 612            be replaced.
 613        """
 614        name = modality.name
 615
 616        if name in self.modalities and not replace:
 617            raise OverwriteError(
 618                "A modality named " + name +
 619                " already exists and replace flag was False.")
 620
 621        if replace:
 622            self.modalities[name] = modality
 623            _logger.debug("Replaced modality %s.", name)
 624        else:
 625            self.modalities[name] = modality
 626            _logger.debug("Added new modality %s.", name)
 627
 628    def after_modalities_init(self) -> None:
 629        """
 630        Ensure everything is properly in place after loading modalities.
 631
 632        Currently, this is only used to create placeholder TextGrids when
 633        needed.
 634        """
 635        textgrid_path = None
 636        if self.recorded_meta_path:
 637            textgrid_path = self.recorded_meta_path.with_suffix(
 638                SourceSuffix.TEXTGRID
 639            )
 640        elif self.recorded_data_path:
 641            textgrid_path = self.recorded_data_path.with_suffix(
 642                SourceSuffix.TEXTGRID
 643            )
 644
 645        if not textgrid_path or not textgrid_path.exists():
 646            if self.file_info.patkit_meta_file:
 647                textgrid_path = self.recorded_path
 648                name = Path(self.file_info.patkit_meta_file)
 649                textgrid_path = stem_path(textgrid_path/name)
 650                textgrid_path = textgrid_path.with_suffix(
 651                    SourceSuffix.TEXTGRID
 652                )
 653
 654        if textgrid_path is not None and textgrid_path.exists():
 655            self.textgrid_path = textgrid_path
 656            self.textgrid = self._read_textgrid()
 657
 658        if self.textgrid is None:
 659            if 'MonoAudio' in self.modalities:
 660                _logger.warning(
 661                    "%s: Creating a placeholder textgrid to make GUI function "
 662                    "correctly.", self.basename)
 663                textgrid = textgrids.TextGrid()
 664                interval = textgrids.Interval(
 665                    text=self.metadata.prompt,
 666                    xmin=self['MonoAudio'].min_time,
 667                    xmax=self['MonoAudio'].max_time,
 668                )
 669                tier = textgrids.Tier(
 670                    xmin=self['MonoAudio'].min_time,
 671                    xmax=self['MonoAudio'].max_time,
 672                )
 673                tier.append(interval)
 674                textgrid['Utterance'] = tier
 675                self.textgrid = textgrid
 676            else:
 677                _logger.warning("No audio found for %s.",
 678                                self.basename)
 679                _logger.warning("Can't create a textgrid so GUI may not "
 680                                "function correctly.")
 681
 682        if self.textgrid:
 683            self.patgrid = PatGrid(self.textgrid)
 684        else:
 685            self.patgrid = None
 686
 687    def __str__(self) -> str:
 688        """
 689        Bare minimum string representation of the Recording.
 690
 691        Returns
 692        -------
 693        string
 694            'Recording [basename]'
 695        """
 696        return f"Recording {self.name}"
 697
 698    def __repr__(self) -> str:
 699        """Overrides the default implementation"""
 700        modalities = "{\n"
 701        for modality in self.modalities:
 702            modalities += ("\t" + modality
 703                           + f": {self.modalities[modality].name},\n")
 704        modalities += "}"
 705        return (
 706            f"Recording {self.basename}\n"
 707            f"{modalities}")
 708
 709    def __eq__(self, other):
 710        """Overrides the default implementation"""
 711        if isinstance(other, Recording):
 712            return self.name == other.name
 713        return NotImplemented

 716class Modality(AbstractData, OrderedDict):
 717    """
 718    Abstract superclass for all data Modality classes.
 719
 720    Any annotations associated with a Modality instance are accessible directly
 721    by `modality[annotation_type]`, because a Modality is also a OrderedDict of
 722    its Annotations.
 723    """
 724    container: Recording
 725
 726    @classmethod
 727    @abc.abstractmethod
 728    def generate_name(cls, params: ModalityMetaData) -> str:
 729        """Abstract version of generating a Modality name."""
 730
 731    def __init__(
 732            self,
 733            container: Recording,
 734            file_info: FileInformation,
 735            parsed_data: ModalityData | None = None,
 736            metadata: ModalityMetaData | None = None,
 737            time_offset: float | None = None,
 738            point_annotations: dict[AnnotationType, PointAnnotations] | None =
 739            None
 740    ) -> None:
 741        """
 742        Modality constructor.
 743
 744        Positional arguments:
